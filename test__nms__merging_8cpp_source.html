22pcod_common::BoundingBox MakeBox(float x, float score, float probability) {
23 pcod_common::BoundingBox box;
24 box.center = {x, 0.0f};
25 box.length = 4.0f;
26 box.width = 2.0f;
27 box.height = 2.0f;
28 box.existence_probability = 0.7f;
29 box.detection_score = score;
30 box.classification = {{0, std::log(probability)}, {1, std::log(1.0f - probability)}};

39double Probability(const pcod_common::BoundingBox& box, std::size_t index) {
40 double denominator = 0.0, numerator = 0.0;
41 float maximum = -std::numeric_limits<float>::infinity();
42 for (const auto& entry : box.classification) {
43 maximum = std::max(maximum, entry.score);
44 }
45 for (const auto& entry : box.classification) {
46 const double value = std::exp(static_cast<double>(entry.score) - maximum);
47 denominator += value;
48 if (entry.class_idx == index) {
49 numerator += value;
50 }
51 }
52 return numerator / denominator;
53}

60std::vector<float> ReferenceNms(std::vector<pcod_common::BoundingBox> boxes, const pcod_common::NmsConfig& config) {
61 std::sort(boxes.begin(), boxes.end(), [](const auto& a, const auto& b) { return *a.detection_score > *b.detection_score; });
62 std::vector<pcod_common::BoundingBox> kept;
63 for (const auto& box : boxes) {
64 bool suppressed = false;
65 const bool class_zero = Probability(box, 0) > 0.5;
66 for (const auto& other : kept) {
67 if (class_zero == (Probability(other, 0) > 0.5) && box.overlaps(other, config.iou_threshold)) {
68 suppressed = true;
69 break;
70 }
71 }
72 if (!suppressed) {
73 kept.push_back(box);
74 if (kept.size() == static_cast<std::size_t>(config.max_detections)) {
75 break;
76 }
77 }
78 }
79 std::vector<float> coordinates;
80 for (const auto& box : kept) {
81 coordinates.push_back(box.center[0]);
82 }
83 return coordinates;
84}
