27Polygon rectangle_vertices(const at::Tensor& box) {
28 // box: [x, y, z, l, w, h, yaw]
29 const float cx = box[0].item<float>();
30 const float cy = box[1].item<float>();
31 const float length = box[3].item<float>();
32 const float width = box[4].item<float>();
33 const float yaw = box[6].item<float>();
34
35 const float dx = length * 0.5f;
36 const float dy = width * 0.5f;
37 const float c = std::cos(yaw);
38 const float s = std::sin(yaw);
39
40 Polygon verts;
41 verts.reserve(4);
42 verts.push_back({cx - dx * c + dy * s, cy - dx * s - dy * c});
43 verts.push_back({cx + dx * c + dy * s, cy + dx * s - dy * c});
44 verts.push_back({cx + dx * c - dy * s, cy + dx * s + dy * c});
45 verts.push_back({cx - dx * c - dy * s, cy - dx * s + dy * c});
46
47 // --- enforce CCW winding ---
48 float area2 = 0.0f;
49 for (int i = 0; i < 4; i++) {
50 const auto& p = verts[i];
51 const auto& q = verts[(i + 1) % 4];
52 area2 += p[0] * q[1] - p[1] * q[0];
53 }
54 if (area2 < 0.0f) {
55 std::reverse(verts.begin(), verts.end());
56 }
57
58 return verts;
59}

61float polygon_area(const Polygon& poly) {
62 if (poly.size() < 3) {
63 return 0.0f;
64 }
65 float area = 0.0f;
66 for (size_t i = 0; i < poly.size(); ++i) {
67 const Point& p = poly[i];
68 const Point& q = poly[(i + 1) % poly.size()];
69 area += cross(p, q);
70 }
71 return 0.5f * std::abs(area);
72}

74Polygon clip_polygon(const Polygon& subject, const Point& edge_start, const Point& edge_end) {
75 Polygon output;
76 if (subject.empty()) {
77 return output;
78 }
79 const Point edge = edge_end - edge_start;
80 const auto eval = [&](const Point& p) {
81 Point v = p - edge_start;
82 return edge[0] * v[1] - edge[1] * v[0]; // cross(edge, v)
83 };
84
85 for (size_t i = 0; i < subject.size(); ++i) {
86 const Point& curr = subject[i];
87 const Point& prev = subject[(i + subject.size() - 1) % subject.size()];
88 float curr_val = eval(curr);
89 float prev_val = eval(prev);
90
91 // For CCW polygons, interior is on the LEFT side => cross >= 0
92 bool curr_inside = curr_val >= 0.0f;
93 bool prev_inside = prev_val >= 0.0f;
94
95 if (curr_inside && prev_inside) {
96 output.push_back(curr);
97 } else if (prev_inside && !curr_inside) {
98 float t = prev_val / (prev_val - curr_val + 1e-12f);
99 Point intersect = prev + (curr - prev) * t;
100 output.push_back(intersect);
101 } else if (!prev_inside && curr_inside) {
102 float t = prev_val / (prev_val - curr_val + 1e-12f);
103 Point intersect = prev + (curr - prev) * t;
104 output.push_back(intersect);
105 output.push_back(curr);
106 }
107 }
108 return output;
109}

111Polygon polygon_intersection(const Polygon& a, const Polygon& b) {
112 Polygon output = a;
113 for (size_t i = 0; i < b.size(); ++i) {
114 if (output.size() < 3) {
115 break;
116 }
117 const Point& p1 = b[i];
118 const Point& p2 = b[(i + 1) % b.size()];
119 output = clip_polygon(output, p1, p2);
120 }
121 return output;
122}

124float oriented_iou(const at::Tensor& box_a, const at::Tensor& box_b) {
125 Polygon poly_a = rectangle_vertices(box_a);
126 Polygon poly_b = rectangle_vertices(box_b);
127 Polygon inter = polygon_intersection(poly_a, poly_b);
128 float inter_area = polygon_area(inter);
129 if (inter_area <= 0.0f) {
130 return 0.0f;
131 }
132 float area_a = polygon_area(poly_a);
133 float area_b = polygon_area(poly_b);
134 float uni = area_a + area_b - inter_area;
135 return inter_area / std::max(uni, 1e-6f);
136}

140// boxes: [N,7], scores: [N], return indices to keep
141torch::Tensor rotated_nms(torch::Tensor boxes, torch::Tensor scores, double iou_threshold, int64_t max_out) {
142 TORCH_CHECK(boxes.dim() == 2 && boxes.size(1) == 7, "boxes must be [N,7]");
143 TORCH_CHECK(scores.dim() == 1 && scores.size(0) == boxes.size(0), "scores must be [N]");
144
145 auto boxes_cpu = boxes.contiguous().to(torch::kCPU);
146 auto scores_cpu = scores.contiguous().to(torch::kCPU);
147 auto order = std::get<1>(scores_cpu.sort(0, /*descending=*/true));
148
149 std::vector<int64_t> keep;
150 keep.reserve(boxes_cpu.size(0));
151
152 for (int64_t oi = 0; oi < order.size(0); ++oi) {
153 int64_t idx = order[oi].item<int64_t>();
154 bool suppress = false;
155 for (int64_t kept_idx : keep) {
156 float iou = oriented_iou(boxes_cpu[idx], boxes_cpu[kept_idx]);
157 if (iou > static_cast<float>(iou_threshold)) {
158 suppress = true;
159 break;
160 }
161 }
162 if (!suppress) {
163 keep.push_back(idx);
164 if (static_cast<int64_t>(keep.size()) >= max_out) {
165 break;
166 }
167 }
168 }
169
170 auto options = torch::TensorOptions().dtype(torch::kLong).device(boxes.device());
171 return torch::from_blob(keep.data(), {static_cast<long>(keep.size())}, options).clone();
172}

174torch::Tensor oriented_iou_aligned(torch::Tensor boxes_a, torch::Tensor boxes_b) {
175 TORCH_CHECK(boxes_a.dim() == 2 && boxes_a.size(1) == 7, "boxes_a must be [N,7]");
176 TORCH_CHECK(boxes_b.dim() == 2 && boxes_b.size(1) == 7, "boxes_b must be [N,7]");
177 TORCH_CHECK(boxes_a.sizes() == boxes_b.sizes(), "boxes_a and boxes_b must have the same shape");
178 auto boxes_a_c = boxes_a.contiguous().to(torch::kCPU);
179 auto boxes_b_c = boxes_b.contiguous().to(torch::kCPU);
180 const auto N = boxes_a_c.size(0);
181 auto out = torch::zeros({N}, boxes_a.options().device(torch::kCPU).dtype(torch::kFloat));
182 for (int64_t i = 0; i < N; ++i) {
183 out[i] = oriented_iou(boxes_a_c[i], boxes_b_c[i]);
184 }
185 return out.to(boxes_a.device());
186}
