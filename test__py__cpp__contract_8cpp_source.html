32std::string Trim(const std::string& value) {
33 std::size_t first = 0;
34 while (first < value.size() && std::isspace(static_cast<unsigned char>(value[first]))) {
35 ++first;
36 }
37 std::size_t last = value.size();
38 while (last > first && std::isspace(static_cast<unsigned char>(value[last - 1]))) {
39 --last;
40 }
41 return value.substr(first, last - first);
42}

48CommandResult RunCommandCapture(const std::string& command) {
49 CommandResult result;
50 FILE* pipe = popen(command.c_str(), "r");
51 if (pipe == nullptr) {
52 return result;
53 }
54 char buffer[256];
55 while (fgets(buffer, static_cast<int>(sizeof(buffer)), pipe) != nullptr) {
56 result.stdout_text += buffer;
57 }
58 result.exit_code = pclose(pipe);
59 return result;
60}

80std::vector<float> ParseCsvFloats(const std::string& csv) {
81 std::vector<float> out;
82 std::stringstream ss(csv);
83 std::string item;
84 while (std::getline(ss, item, ',')) {
85 const std::string trimmed = Trim(item);
86 if (!trimmed.empty()) {
87 out.push_back(std::stof(trimmed));
