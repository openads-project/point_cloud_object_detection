225std::optional<nav_msgs::msg::OccupancyGrid> initializeAuxiliaryOccupancyGridMessage(const AuxiliaryGridMap& grid_map_output,
226 const std_msgs::msg::Header& header,
227 const std::string& frame_id,
228 const geometry_msgs::msg::Pose& origin) {
229 if (grid_map_output.grid_x <= 0 || grid_map_output.grid_y <= 0) {
230 return std::nullopt;
231 }
232 if (grid_map_output.values.size() != static_cast<std::size_t>(grid_map_output.grid_x * grid_map_output.grid_y)) {
233 return std::nullopt;
234 }
235
236 const double x_length = static_cast<double>(grid_map_output.x_max) - static_cast<double>(grid_map_output.x_min);
237 const double y_length = static_cast<double>(grid_map_output.y_max) - static_cast<double>(grid_map_output.y_min);
238 if (!(x_length > 0.0) || !(y_length > 0.0)) {
239 return std::nullopt;
240 }
241
242 const double resolution_x = x_length / static_cast<double>(grid_map_output.grid_x);
243 const double resolution_y = y_length / static_cast<double>(grid_map_output.grid_y);
244 if (std::fabs(resolution_x - resolution_y) > kAuxiliaryGridMapResolutionTolerance) {
245 return std::nullopt;
246 }
247
248 nav_msgs::msg::OccupancyGrid message;
249 message.header = header;
250 message.header.frame_id = frame_id;
251 message.info.map_load_time = header.stamp;
252 message.info.resolution = static_cast<float>(resolution_x);
253 message.info.width = static_cast<std::uint32_t>(grid_map_output.grid_x);
254 message.info.height = static_cast<std::uint32_t>(grid_map_output.grid_y);
255 message.info.origin = origin;
256 message.data.assign(static_cast<std::size_t>(grid_map_output.grid_x * grid_map_output.grid_y), 0);
257 return message;
258}

264std::optional<nav_msgs::msg::OccupancyGrid> buildAuxiliaryOccupancyGridMessage(const AuxiliaryGridMap& grid_map_output,
265 const std_msgs::msg::Header& header,
266 const std::string& frame_id,
267 const geometry_msgs::msg::Pose& origin) {
268 auto message = initializeAuxiliaryOccupancyGridMessage(grid_map_output, header, frame_id, origin);
269 if (!message.has_value()) {
270 return std::nullopt;
271 }
272
273 for (int ix = 0; ix < grid_map_output.grid_x; ++ix) {
274 for (int iy = 0; iy < grid_map_output.grid_y; ++iy) {
275 const std::size_t source_index = static_cast<std::size_t>(ix * grid_map_output.grid_y + iy);
276 const std::size_t target_index = static_cast<std::size_t>(iy * grid_map_output.grid_x + ix);
277 message->data[target_index] = toOccupancyGridCellValue(grid_map_output.values[source_index]);
278 }
279 }
280
281 return message;
282}

301std::optional<geometry_msgs::msg::Pose> transformAuxiliaryGridMapOriginPose(const geometry_msgs::msg::Pose& source_origin,
302 const std_msgs::msg::Header& header,
303 const std::string& source_frame,
304 const std::string& target_frame,
305 tf2_ros::Buffer& tf_buffer,
306 const rclcpp::Logger& logger) {
307 if (target_frame.empty() || target_frame == source_frame) {
308 return source_origin;
309 }
310
311 geometry_msgs::msg::TransformStamped transform;
312 try {
313 transform = tf_buffer.lookupTransform(target_frame, source_frame, header.stamp);
314 } catch (tf2::TransformException& e) {
315 RCLCPP_ERROR(logger,
316 "Cannot transform grid map: Transformation from source frame "
317 "(%s) to output frame (%s) not found: %s",
318 source_frame.c_str(), target_frame.c_str(), e.what());
319 return std::nullopt;
320 }
321
322 const auto& translation = transform.transform.translation;
323 const auto& rotation = transform.transform.rotation;
324 tf2::Quaternion target_from_source_rotation(rotation.x, rotation.y, rotation.z, rotation.w);
325 target_from_source_rotation.normalize();
326 const tf2::Transform target_from_source(target_from_source_rotation, tf2::Vector3(translation.x, translation.y, translation.z));
327
328 const auto& source_position = source_origin.position;
329 const auto& source_orientation = source_origin.orientation;
330 tf2::Quaternion source_origin_rotation(source_orientation.x, source_orientation.y, source_orientation.z, source_orientation.w);
331 source_origin_rotation.normalize();
332 const tf2::Transform source_from_grid(source_origin_rotation,
333 tf2::Vector3(source_position.x, source_position.y, source_position.z));
334
335 const tf2::Transform target_from_grid = target_from_source * source_from_grid;
336
337 geometry_msgs::msg::Pose target_origin;
338 const tf2::Vector3 target_position = target_from_grid.getOrigin();
339 target_origin.position.x = target_position.x();
340 target_origin.position.y = target_position.y();
341 target_origin.position.z = target_position.z();
342 const tf2::Quaternion target_orientation = target_from_grid.getRotation();
343 target_origin.orientation.x = target_orientation.x();
344 target_origin.orientation.y = target_orientation.y();
345 target_origin.orientation.z = target_orientation.z();
346 target_origin.orientation.w = target_orientation.w();
347 return target_origin;
348}

350std::optional<AuxiliaryGridMap> buildCombinedAuxiliaryGridMap(const AuxiliaryGridMap& density_grid_map,
351 const AuxiliaryGridMap& dynamic_grid_map) {
352 if (!haveCompatibleAuxiliaryGridMaps(density_grid_map, dynamic_grid_map)) {
353 return std::nullopt;
354 }
355
356 AuxiliaryGridMap combined_grid_map = makeAuxiliaryGridMapLike(density_grid_map, "combined");
357 for (std::size_t i = 0; i < combined_grid_map.values.size(); ++i) {
358 const float density = clamp(density_grid_map.values[i], 0.0F, 1.0F);
359 const float dynamic_occupancy = clamp(dynamic_grid_map.values[i], 0.0F, 1.0F);
360 combined_grid_map.values[i] = dynamic_occupancy + 0.5F * density * (1.0F - dynamic_occupancy);
361 }
362 return combined_grid_map;
363}

391constexpr double kGridMapMaskMinDetectionAreaRadius = 0.0;
392constexpr double kGridMapMaskMinDetectionAreaFovDeg = 0.0;
393constexpr double kGridMapMaskMaxDetectionAreaFovDeg = 360.0;
394
395bool isPointInsideDetectionAreaSector(double x, double y, const Params& params) {
396 const double sector_center_x = params.detection_area_center_x;
397 const double sector_center_y = params.detection_area_center_y;
398 const double sector_radius = params.detection_area_radius;
399 const double sector_bearing_rad = params.detection_area_bearing_deg * M_PI / 180.0;
400 const double sector_fov_rad = params.detection_area_fov_deg * M_PI / 180.0;
401 constexpr double kFullCircleToleranceDeg = 1e-3;
402 const bool is_full_circle =
403 std::abs(params.detection_area_fov_deg - kGridMapMaskMaxDetectionAreaFovDeg) <= kFullCircleToleranceDeg;
404
405 const double delta_x = x - sector_center_x;
406 const double delta_y = y - sector_center_y;
407 const double squared_distance_to_center = delta_x * delta_x + delta_y * delta_y;
408 if (squared_distance_to_center > sector_radius * sector_radius) {
409 return false;
410 }
411 if (is_full_circle) {
412 return true;
413 }
414
415 const double angle_to_center = std::atan2(delta_y, delta_x);
416 double angle_offset = angle_to_center - sector_bearing_rad;
417 while (angle_offset > M_PI) angle_offset -= 2.0 * M_PI;
418 while (angle_offset < -M_PI) angle_offset += 2.0 * M_PI;
419 return std::abs(angle_offset) <= (sector_fov_rad * 0.5 + 1e-9);
420}

422void applyAuxiliaryGridMapMasks(AuxiliaryGridMap& grid_map_output, const Params& params) {
423 if (grid_map_output.grid_x <= 0 || grid_map_output.grid_y <= 0 || grid_map_output.values.empty()) {
424 return;
425 }
426
427 const bool zero_in_no_detection_zone = params.zero_grid_map_cells_in_no_detection_zone && params.no_detection_zone_enabled;
428 const bool zero_outside_detection_area = params.zero_grid_map_cells_outside_detection_area && params.detection_area_enabled &&
429 params.detection_area_radius > kGridMapMaskMinDetectionAreaRadius &&
430 params.detection_area_fov_deg > kGridMapMaskMinDetectionAreaFovDeg;
431 if (!zero_in_no_detection_zone && !zero_outside_detection_area) {
432 return;
433 }
434
435 const double resolution_x =
436 (static_cast<double>(grid_map_output.x_max) - static_cast<double>(grid_map_output.x_min)) / grid_map_output.grid_x;
437 const double resolution_y =
438 (static_cast<double>(grid_map_output.y_max) - static_cast<double>(grid_map_output.y_min)) / grid_map_output.grid_y;
439
440 for (int ix = 0; ix < grid_map_output.grid_x; ++ix) {
441 for (int iy = 0; iy < grid_map_output.grid_y; ++iy) {
442 const double x = static_cast<double>(grid_map_output.x_min) + (static_cast<double>(ix) + 0.5) * resolution_x;
443 const double y = static_cast<double>(grid_map_output.y_min) + (static_cast<double>(iy) + 0.5) * resolution_y;
444
445 const bool in_no_detection_zone = zero_in_no_detection_zone && x >= params.no_detection_zone_x_min &&
446 x <= params.no_detection_zone_x_max && y >= params.no_detection_zone_y_min &&
447 y <= params.no_detection_zone_y_max;
448 const bool outside_detection_area = zero_outside_detection_area && !isPointInsideDetectionAreaSector(x, y, params);
449
450 if (in_no_detection_zone || outside_detection_area) {
451 const std::size_t flat_index = static_cast<std::size_t>(ix * grid_map_output.grid_y + iy);
452 grid_map_output.values[flat_index] = 0.0F;
453 }
454 }
455 }

470PreparedPointCloudInput preparePointCloudInput(const sensor_msgs::msg::PointCloud2::ConstSharedPtr& msg,
471 const ModelConfig& model_config,
472 const Params& params,
473 tf2_ros::Buffer& tf_buffer,
474 const rclcpp::Logger& logger) {
475 PreparedPointCloudInput prepared;
476 prepared.msg = msg;
477
478 if (!params.inference_frame.empty() && msg->header.frame_id != params.inference_frame) {
479 prepared.transformed_owner = std::make_shared<sensor_msgs::msg::PointCloud2>();
480 try {
481 tf_buffer.transform(*msg, *prepared.transformed_owner, params.inference_frame);
482 prepared.msg = prepared.transformed_owner;
483 } catch (tf2::TransformException& e) {
484 RCLCPP_ERROR(logger,
485 "Cannot tranform Pointcloud: Transformation from its frame "
486 "(%s) to inference_frame "
487 "(%s) not found: %s",
488 msg->header.frame_id.c_str(), params.inference_frame.c_str(), e.what());
489 throw;
490 }
491 }
492
493 auto findField = [&](const std::string& name) -> const sensor_msgs::msg::PointField* {
494 for (const auto& field : prepared.msg->fields) {
495 if (field.name == name) {
496 return &field;
497 }
498 }
499 return nullptr;
500 };
501
502 const std::string primary_feature_field = resolvePrimaryFeatureField(model_config, params);
503 const auto* x_field = findField("x");
504 const auto* y_field = findField("y");
505 const auto* z_field = findField("z");
506 const auto* primary_feature = findField(primary_feature_field);
507
508 if ((x_field == nullptr) || (y_field == nullptr) || (z_field == nullptr)) {
509 RCLCPP_FATAL(logger, "Point fields 'x', 'y', 'z' are required in PointCloud2 input");
510 throw std::runtime_error("Missing required PointCloud2 coordinate fields");
511 }
512 if (primary_feature == nullptr) {
513 RCLCPP_FATAL(logger, "Point field '%s' is required for primary feature extraction", primary_feature_field.c_str());
514 throw std::runtime_error("Missing required PointCloud2 field: " + primary_feature_field);
515 }
516
517 using PF = sensor_msgs::msg::PointField;
518 if (x_field->datatype != PF::FLOAT32 || y_field->datatype != PF::FLOAT32 || z_field->datatype != PF::FLOAT32) {
519 RCLCPP_FATAL(logger, "Point fields 'x', 'y', 'z' must be FLOAT32");
520 throw std::runtime_error("Unsupported PointCloud2 datatype for x/y/z fields");
521 }
522
523 switch (primary_feature->datatype) {
524 case PF::FLOAT32:
525 break;
526 case PF::FLOAT64:
527 case PF::UINT16:
528 case PF::UINT8:
529 case PF::INT16:
530 case PF::INT8:
531 case PF::UINT32:
532 case PF::INT32:
533 RCLCPP_WARN_ONCE(logger, "Converting PointCloud2 field '%s' from %s to FLOAT32", primary_feature_field.c_str(),
534 pointFieldDatatypeToString(primary_feature->datatype));
535 break;
536 default:
537 RCLCPP_FATAL(logger,
538 "Point field '%s' has unsupported datatype %u (%s). "
539 "Supported: INT8, UINT8, INT16, UINT16, "
540 "INT32, UINT32, FLOAT32, FLOAT64",
541 primary_feature_field.c_str(), primary_feature->datatype,
542 pointFieldDatatypeToString(primary_feature->datatype));
543 throw std::runtime_error("Unsupported PointCloud2 datatype for field: " + primary_feature_field);
544 }
545
546 prepared.x_offset = x_field->offset;
547 prepared.y_offset = y_field->offset;
548 prepared.z_offset = z_field->offset;
549 prepared.feature_offset = primary_feature->offset;
550 prepared.feature_datatype = primary_feature->datatype;
551 prepared.needs_swap = prepared.msg->is_bigendian == isHostLittleEndian();
552 prepared.total_points = static_cast<std::size_t>(prepared.msg->width) * prepared.msg->height;
553 return prepared;
554}

556void decodePreparedPointCloudToPcl(const PreparedPointCloudInput& prepared, PointCloud& point_cloud) {
557 const auto& msg_ref = *prepared.msg;
558 point_cloud.clear();
559 pcl_conversions::toPCL(msg_ref.header, point_cloud.header);
560 point_cloud.width = msg_ref.width;
561 point_cloud.height = msg_ref.height;
562 point_cloud.is_dense = msg_ref.is_dense;
563 point_cloud.points.resize(prepared.total_points);
564
565 std::size_t out_idx = 0;
566 for (std::size_t row = 0; row < msg_ref.height; ++row) {
567 for (std::size_t col = 0; col < msg_ref.width; ++col) {
568 const std::size_t point_index = row * static_cast<std::size_t>(msg_ref.width) + col;
569 const PointCloud2PointRecord point_record =
570 readPointCloud2PointRecord(msg_ref, point_index, prepared.x_offset, prepared.y_offset, prepared.z_offset,
571 prepared.feature_offset, prepared.feature_datatype, prepared.needs_swap);
572 point_cloud.points[out_idx++] = makePoint(point_record.x, point_record.y, point_record.z, point_record.feature);
573 }
574 }
575}

1357bool PointCloudObjectDetection::updateNMSScoreThreshold(std::vector<double>& score_thresholds,
1358 const std::vector<std::string>& predicted_class_names) const {
1359 if (score_thresholds.size() == 1) {
1360 for (std::size_t i = 1; i < predicted_class_names.size(); i++) {
1361 // If a single NMS score threshold is provided, replicate it for each
1362 // class.
1363 score_thresholds.push_back(score_thresholds[0]);
1364 }
1365 }
1366 if (score_thresholds.size() != predicted_class_names.size()) {
1367 RCLCPP_ERROR(this->get_logger(),
1368 "NMS score threshold count mismatch: got %zu value(s), "
1369 "expected 1 or %zu (one per predicted class).",
1370 score_thresholds.size(), predicted_class_names.size());
1371 return false;
1372 }
1373 return true;

1479 throwParameterError(this->get_logger(), "postprocessing.nms.max_num_objects", "cannot be negative");
1480 }
1481 if (!updateNMSScoreThreshold(model_config.nms_score_threshold, model_config.predicted_class_names)) {

1611rcl_interfaces::msg::SetParametersResult PointCloudObjectDetection::parametersCallback(
1612 const std::vector<rclcpp::Parameter>& parameters) {
1613 rcl_interfaces::msg::SetParametersResult result;
1614 result.successful = true;
1615 result.reason = "success";
1616 const Params params_before_update = params_;
1617 const ModelConfig model_config_before_update = model_config_;
1618 const pcod_common::NmsConfig nms_config_before_update = nms_config_;
1619 auto rollback_state = [&]() {
1620 params_ = params_before_update;
1621 model_config_ = model_config_before_update;
1622 nms_config_ = nms_config_before_update;
1623 syncModelRuntimeConfigFromParams();
1624 };
1625
1626 const auto name_in = [](const std::string& name, std::initializer_list<const char*> allowed) {
1627 return std::any_of(allowed.begin(), allowed.end(), [&](const char* s) { return name == s; });
1628 };
1629
1630 bool model_change_on_runtime = false;
1631 bool publishers_changed = false;
1632 bool repository_changed = false;
1633 for (const auto& param : parameters) {
1634 if (requiresModelReinitializationForParameter(param.get_name())) {
1635 model_change_on_runtime = true;
1636 }
1637 if (name_in(param.get_name(),
1638 {"preprocessing.no_detection_zone.publish_polygon", "preprocessing.detection_area.publish_polygon",
1639 "preprocessing.no_detection_zone.publish_points", "output.model_bounds.publish_polygon",
1640 "output.grid_maps.publish_density", "output.grid_maps.publish_dynamic", "output.grid_maps.publish_combined",
1641 "output.grid_maps.publish_static"})) {
1642 publishers_changed = true;
1643 }
1644 }
1645
1646 std::vector<rclcpp::Parameter> applied_parameters;
1647
1648 {
1649 std::lock_guard<std::mutex> model_lock(model_mutex_);
1650 try {
1651 for (const auto& param : parameters) {
1652 if (param.get_name() == "prediction.model_repository") {
1653 params_.model_repository = param.as_string();
1654 applied_parameters.push_back(param);
1655 repository_changed = (params_.model_repository != params_before_update.model_repository);
1656 continue;
1657 }
1658 if (param.get_name() == "prediction.model_version") {
1659 params_.model_version = param.as_string();
1660 applied_parameters.push_back(param);
1661 continue;
1662 }
1663 for (auto& auto_reconfigurable_param : auto_reconfigurable_params_) {
1664 if (param.get_name() == std::get<0>(auto_reconfigurable_param)) {
1665 std::get<1>(auto_reconfigurable_param)(param);
1666 applied_parameters.push_back(param);
1667 break;
1668 }
1669 }
1670 }
1671
1672 const double cscu = pm::object_access::CONTINUOUS_STATE_COVARIANCE_UNKNOWN;
1673 sanitizeVarianceVector(params_.variance, cscu);
1674 if (repository_changed) {
1675 // All runtime-configurable fields are read from the current params_
1676 // inside refreshResolvedModelConfigLocked, so they are preserved across
1677 // the model switch.
1678 refreshResolvedModelConfigLocked();
1679 } else {
1680 syncModelRuntimeConfigFromParams();
1681 syncNmsRuntimeConfigFromParams();
1682 }
1683 validateParamsOrThrow();
1684 if (detection_model_) {
1685 validateModelConfigOrThrow();
1686 }
1687 } catch (const std::exception& e) {
1688 rollback_state();
1689 result.successful = false;
1690 result.reason = std::string("Invalid parameter update: ") + e.what();
1691 RCLCPP_ERROR(this->get_logger(), "%s", result.reason.c_str());
1692 return result;
1693 } catch (...) {
1694 rollback_state();
1695 result.successful = false;
1696 result.reason = "Invalid parameter update: unknown validation error";
1697 RCLCPP_ERROR(this->get_logger(), "%s", result.reason.c_str());
1698 return result;
1699 }
1700 }
1701
1702 for (const auto& param : applied_parameters) {
1703 RCLCPP_INFO(this->get_logger(), "Reconfigured parameter '%s' to: %s", param.get_name().c_str(),
1704 param.value_to_string().c_str());
1705 }
1706
1707 // Reinitialize model if runtime-critical configuration changed
1708 if (model_change_on_runtime) {
1709 try {
1710 model_ready_.store(false, std::memory_order_release);
1711 initializeModel();
1712 RCLCPP_INFO(this->get_logger(), "Successfully reinitialized model with name: %s, version: %s", params_.model_name.c_str(),
1713 params_.model_version.c_str());
1714 } catch (const std::exception& e) {
1715 {
1716 std::lock_guard<std::mutex> model_lock(model_mutex_);
1717 rollback_state();
1718 model_ready_.store(detection_model_ != nullptr, std::memory_order_release);
1719 }
1720 result.successful = false;
1721 result.reason = "Failed to reinitialize model: " + std::string(e.what());
1722 RCLCPP_ERROR(this->get_logger(), "Failed to reinitialize model: %s", e.what());
1723 return result;
1724 }
1725 }
1726
1727 // Defer publisher reconfiguration to normal execution context.
1728 // Some transport publishers may declare/set parameters internally, which is
1729 // forbidden in set-parameter callbacks.
1730 if (publishers_changed) {
1731 publishers_update_pending_.store(true, std::memory_order_release);
1732 }
1733
1734 return result;
1735}

1748void PointCloudObjectDetection::initializeModel() {
1749 std::lock_guard<std::mutex> model_lock(model_mutex_);
1750 if (params_.model_name.empty() || params_.model_version.empty() || model_config_.predicted_class_names.empty()) {
1751 refreshResolvedModelConfigLocked();
1752 }
1753
1754 const Params params_snapshot = params_;
1755 const std::string model_name = params_.model_name;
1756 const std::string model_version = params_.model_version;
1757 ModelConfig new_model_config = model_config_;
1758 std::unique_ptr<triton_cpp::TritonInterface> new_triton_interface;
1759 std::unique_ptr<Model> new_detection_model;
1760 if (params_snapshot.cuda_input_shm && model_config_.preprocessing_backend != "cuda") {
1761 RCLCPP_WARN(this->get_logger(),
1762 "prediction.cuda_input_shm is enabled but preprocessing.backend='%s'. "
1763 "CUDA input shared memory is only used with "
1764 "preprocessing.backend='cuda'; "
1765 "continuing with the normal CPU input transport.",
1766 model_config_.preprocessing_backend.c_str());
1767 }
1768
1769 // Retry Triton connection until the ROS context is shutting down.
1770 constexpr auto kRetryDelay = std::chrono::seconds(1);
1771 while (rclcpp::ok(this->get_node_base_interface()->get_context())) {
1772 try {
1773 new_triton_interface = std::make_unique<triton_cpp::TritonInterface>(
1774 model_name, model_version, params_snapshot.server_url, params_snapshot.use_shm, false, false,
1775 params_snapshot.triton_client_timeout_s,
1776 params_snapshot.cuda_input_shm && model_config_.preprocessing_backend == "cuda");
1777 break;
1778 } catch (const std::exception& e) {
1779 RCLCPP_WARN(this->get_logger(),
1780 "Failed to connect to Triton server '%s' for model '%s:%s': "
1781 "%s. Retrying in %.1fs.",
1782 params_snapshot.server_url.c_str(), model_name.c_str(), model_version.c_str(), e.what(),
1783 std::chrono::duration<double>(kRetryDelay).count());
1784 rclcpp::sleep_for(kRetryDelay);
1785 }
1786 }
1787
1788 if (!new_triton_interface) {
1789 throw std::runtime_error("Shutdown requested while waiting for Triton connection");
1790 }
1791
1792 // log model info
1793 const std::string model_info = new_triton_interface->getModelInfo();
1794 std::cout << model_info << std::endl;
1795
1796 // create model architecture
1797 PBODModel::validateInterface(*new_triton_interface);
1798 new_detection_model = std::make_unique<PBODModel>(*new_triton_interface, new_model_config);
1799 new_triton_interface->initInOutputs(new_detection_model->getSpecialOutputShapes());
1800 triton_interface_ = std::move(new_triton_interface);
1801 detection_model_ = std::move(new_detection_model);
1802 model_ready_.store(true, std::memory_order_release);
1803}

1942void PointCloudObjectDetection::boxesToObjectList(const std::vector<BoundingBox>& bboxes,
1943 const ModelConfig& model_config,
1944 const Params& params,
1945 pm::msg::ObjectList& object_list) {
1946 // iterate over all boxes
1947 for (int idx = 0; idx < static_cast<int>(bboxes.size()); ++idx) {
1948 pm::msg::Object object;
1949 pm::object_access::initializeState(object, pm::msg::ISCACTR::MODEL_ID);
1950
1951 // Set sensor ID
1952 object.state.sensor_id.push_back(params.sensor_id);
1953
1954 // set id
1955 object.id = idx;
1956
1957 // set occupancy probability
1958 object.existence_probability = bboxes[idx].existence_probability;
1959
1960 // set object position
1961 pm::object_access::setX(object, bboxes[idx].center[0]);
1962 pm::object_access::setY(object, bboxes[idx].center[1]);
1963 pm::object_access::setZ(object, bboxes[idx].z);
1964 pm::object_access::setYaw(object, bboxes[idx].yaw);
1965
1966 // set object dimensions
1967 pm::object_access::setLength(object, bboxes[idx].length);
1968 pm::object_access::setWidth(object, bboxes[idx].width);
1969 pm::object_access::setHeight(object, bboxes[idx].height);
1970
1971 // set velocity
1972 if (bboxes[idx].has_velocity) {
1973 geometry_msgs::msg::Vector3 velocity;
1974 velocity.x = bboxes[idx].v_x;
1975 velocity.y = bboxes[idx].v_y;
1976 velocity.z = 0.0;
