28float sigmoid(float value) {
29 if (value >= 0.0F) {
30 const float z = std::exp(-value);
31 return 1.0F / (1.0F + z);
32 }
33 const float z = std::exp(value);
34 return z / (1.0F + z);
35}

37float densityTransform(float value) {
38 const float normalized = std::asinh(kDensityTransformScale * value) / std::asinh(kDensityTransformScale);
39 return std::clamp(normalized, 0.0F, 1.0F);
40}

44void PBODModel::validateInterface(const triton_cpp::TritonInterface& triton_interface) {
45 if (triton_interface.nInputs() != kExpectedInputNames.size()) {
46 throw std::runtime_error("PBOD model interface mismatch: expected " + std::to_string(kExpectedInputNames.size()) +
47 " inputs but Triton reports " + std::to_string(triton_interface.nInputs()));
48 }
49
50 for (const char* input_name : kExpectedInputNames) {
51 try {
52 (void)triton_interface.getInputShape(input_name);
53 } catch (const std::invalid_argument& e) {
54 throw std::runtime_error("PBOD model is missing expected input tensor '" + std::string(input_name) + "': " + e.what());
55 }
56 }
57
58 for (const char* output_name : {kOutputNameFocal, kOutputNameReg, kOutputNameClass, kOutputNameSize}) {
59 try {
60 (void)triton_interface.getOutputShape(output_name);
61 } catch (const std::invalid_argument& e) {
62 throw std::runtime_error("PBOD model is missing expected output tensor '" + std::string(output_name) + "': " + e.what());
63 }
64 }
65
66 bool has_density = true;
67 bool has_dynamic_occupancy = true;
68 try {
69 (void)triton_interface.getOutputShape(kOutputNameDensity);
70 } catch (const std::invalid_argument&) {
71 has_density = false;
72 }
73 try {
74 (void)triton_interface.getOutputShape(kOutputNameDynamicOccupancy);
75 } catch (const std::invalid_argument&) {
76 has_dynamic_occupancy = false;
77 }
78 if (has_density != has_dynamic_occupancy) {
79 throw std::runtime_error(
80 "PBOD model must expose both density_logits and "
81 "dynamic_occupancy_logits together.");
82 }
83}

121 z_min_,

172 pillar_ids_raw_.resize(static_cast<std::size_t>(max_num_points_), -1);
173 pillar_ix_raw_.resize(static_cast<std::size_t>(max_num_points_), -1);
174 pillar_iy_raw_.resize(static_cast<std::size_t>(max_num_points_), -1);
175 tensor_reset_state_.active_point_rows.reserve(static_cast<std::size_t>(max_num_points_));
176 tensor_reset_state_.active_pillar_ids.reserve(static_cast<std::size_t>(num_pillars_));
177 active_pillar_ids_scratch_.reserve(static_cast<std::size_t>(num_pillars_));
178 selected_point_records_.reserve(static_cast<std::size_t>(max_num_points_));
179 try {
180 (void)triton_interface_.getOutputShape(kOutputNameDensity);
181 (void)triton_interface_.getOutputShape(kOutputNameDynamicOccupancy);
182 has_auxiliary_grid_map_outputs_ = true;

184 has_auxiliary_grid_map_outputs_ = false;

224template <typename PointGetter>
225void PBODModel::setupModelInputFromGetter(int n_cloud_points, PointGetter get_point, bool materialize_filtered_points) {
226 const bool use_cuda_input_shm = useCudaPreprocessing(model_config_) && triton_interface_.usesCudaInputSharedMemory();
227
228 float* point_features_device = nullptr;
229 std::int64_t* pillar_ids_device = nullptr;
230 bool* valid_mask_device = nullptr;
231 bool* pillar_masks_device = nullptr;
232 float* point_features_host = nullptr;
233 std::int64_t* pillar_ids_host = nullptr;
234 bool* valid_mask_host = nullptr;
235 bool* pillar_masks_host = nullptr;
236
237 if (use_cuda_input_shm) {
238 point_features_device =
239 static_cast<float*>(static_cast<void*>(triton_interface_.getInputTensorDevice(kInputNamePointFeatures).first));
240 pillar_ids_device =
241 static_cast<std::int64_t*>(static_cast<void*>(triton_interface_.getInputTensorDevice(kInputNamePillarIds).first));
242 valid_mask_device = static_cast<bool*>(static_cast<void*>(triton_interface_.getInputTensorDevice(kInputNameValidMask).first));
243 pillar_masks_device =
244 static_cast<bool*>(static_cast<void*>(triton_interface_.getInputTensorDevice(kInputNamePillarMasks).first));
245 if (!pillar_indices_initialized_) {
246 triton_interface_.copyInputTensorToDevice(kInputNamePillarIndices, pillar_indices_.data(),
247 pillar_indices_.size() * sizeof(std::int64_t));
248 pillar_indices_initialized_ = true;
249 }
250 } else {
251 auto point_features_map =
252 triton_interface_.getInputTensor<float>(kInputNamePointFeatures, max_num_points_, preprocessed_feature_dim_);
253 auto pillar_ids_map = triton_interface_.getInputTensor<int64_t>(kInputNamePillarIds, max_num_points_);
254 auto valid_mask_map = triton_interface_.getInputTensor<bool>(kInputNameValidMask, max_num_points_);
255 auto pillar_masks_map = triton_interface_.getInputTensor<bool>(kInputNamePillarMasks, num_pillars_);
256 auto pillar_indices_map = triton_interface_.getInputTensor<int64_t>(kInputNamePillarIndices, num_pillars_, kPillarIndexDim);
257 point_features_host = point_features_map.data();
258 pillar_ids_host = pillar_ids_map.data();
259 valid_mask_host = valid_mask_map.data();
260 pillar_masks_host = pillar_masks_map.data();
261
262 if (!pillar_indices_initialized_) {
263 for (int i = 0; i < num_pillars_; ++i) {
264 const std::size_t idx = static_cast<std::size_t>(i) * kPillarIndexDim;
265 pillar_indices_map(i, 0) = pillar_indices_[idx];
266 pillar_indices_map(i, 1) = pillar_indices_[idx + 1];
267 }
268 pillar_indices_initialized_ = true;
269 }
270 }
271
272 selected_point_records_.clear();
273 selected_point_records_.reserve(std::min(n_cloud_points, max_num_points_));
274 filtered_input_point_count_ = 0;
275 if (materialize_filtered_points) {
276 filtered_input_points_.clear();
277 filtered_input_points_.reserve(std::min(n_cloud_points, max_num_points_));
278 } else {
279 filtered_input_points_.clear();
280 }
281
282 if (n_cloud_points == 0) {
283 RCLCPP_DEBUG(rclcpp::get_logger("PBODModel"), "Input point cloud is empty.");
284 }
285
286 static thread_local std::mt19937 gen(std::random_device{}());
287 static thread_local std::uniform_int_distribution<int> reservoir_dist;
288 int valid_point_count = 0;
289 for (int i = 0; i < n_cloud_points; ++i) {
290 const PointRecord point = get_point(i);
291 if (!point_preprocessor_.IsPointValid(point.x, point.y, point.z)) {
292 continue;
293 }
294 ++valid_point_count;
295 if (valid_point_count <= max_num_points_) {
296 selected_point_records_.push_back(point);
297 if (materialize_filtered_points) {
298 filtered_input_points_.push_back(makePoint(point.x, point.y, point.z, point.intensity));
299 }
300 } else {
301 reservoir_dist.param(std::uniform_int_distribution<int>::param_type(1, valid_point_count));
302 const int random_idx = reservoir_dist(gen);
303 if (random_idx <= max_num_points_) {
304 const std::size_t replace_idx = static_cast<std::size_t>(random_idx - 1);
305 selected_point_records_[replace_idx] = point;
306 if (materialize_filtered_points) {
307 filtered_input_points_[replace_idx] = makePoint(point.x, point.y, point.z, point.intensity);
308 }
309 }
310 }
311 }
312
313 const int num_selected = static_cast<int>(selected_point_records_.size());
314 filtered_input_point_count_ = static_cast<std::size_t>(num_selected);
315
316 if (normalization_type_ == pcod_common::PointFeatureNormalizationType::kZScore) {
317 double sum = 0.0;
318 double sum_sq = 0.0;
319 for (const auto& point : selected_point_records_) {
320 const double value = static_cast<double>(point.intensity);
321 sum += value;
322 sum_sq += value * value;
323 }
324 const double count = static_cast<double>(std::max(1, num_selected));
325 const double mean = sum / count;
326 const double variance = std::max(0.0, sum_sq / count - mean * mean);
327 const double stddev = std::sqrt(variance);
328 point_preprocessor_.SetZScoreStats(static_cast<float>(mean), static_cast<float>(stddev));
329 }
330
331 if (!use_cuda_input_shm) {
332 const auto resetAllTensors = [&]() {
333 auto point_features_map =
334 triton_interface_.getInputTensor<float>(kInputNamePointFeatures, max_num_points_, preprocessed_feature_dim_);
335 auto pillar_ids_map = triton_interface_.getInputTensor<int64_t>(kInputNamePillarIds, max_num_points_);
336 auto valid_mask_map = triton_interface_.getInputTensor<bool>(kInputNameValidMask, max_num_points_);
337 auto pillar_masks_map = triton_interface_.getInputTensor<bool>(kInputNamePillarMasks, num_pillars_);
338 point_features_map.setZero();
339 valid_mask_map.setZero();
340 pillar_masks_map.setZero();
341 pillar_ids_map.setConstant(pillar_id_sentinel_);
342 };
343
344 const auto resetSparseTensors = [&]() {
345 FeatureTensorView point_features_view(point_features_host, max_num_points_, preprocessed_feature_dim_);
346 BoolTensorView valid_mask_view(valid_mask_host, max_num_points_);
347 Int64TensorView pillar_ids_view(pillar_ids_host, max_num_points_);
348 BoolTensorView pillar_masks_view(pillar_masks_host, num_pillars_);
349
350 if (!tensor_reset_state_.initialized) {
351 resetAllTensors();
352 tensor_reset_state_.initialized = true;
353 tensor_reset_state_.active_point_rows.clear();
354 tensor_reset_state_.active_pillar_ids.clear();
355 return;
356 }
357
358 for (int row : tensor_reset_state_.active_point_rows) {
359 for (int feature_idx = 0; feature_idx < preprocessed_feature_dim_; ++feature_idx) {
360 point_features_view(row, feature_idx) = 0.0F;
361 }
362 valid_mask_view(row) = false;
363 pillar_ids_view(row) = pillar_id_sentinel_;
364 }
365 for (int pillar_id : tensor_reset_state_.active_pillar_ids) {
366 pillar_masks_view(pillar_id) = false;
367 }
368 tensor_reset_state_.active_point_rows.clear();
369 tensor_reset_state_.active_pillar_ids.clear();
370 };
371 resetSparseTensors();
372 } else {
373 tensor_reset_state_.active_point_rows.clear();
374 tensor_reset_state_.active_pillar_ids.clear();
375 }
376
377 if (use_cuda_input_shm) {
378 if (populateModelInputOnGpuToDevice(point_features_device, pillar_ids_device, valid_mask_device, pillar_masks_device,
379 num_selected)) {
380 return;
381 }
382 throw std::runtime_error(
383 "Failed to populate PBOD input tensors via CUDA shared memory after "
384 "prediction.cuda_input_shm was enabled");
385 }
386
387 if (useCudaPreprocessing(model_config_) &&
388 populateModelInputOnGpu(point_features_host, pillar_ids_host, valid_mask_host, pillar_masks_host, num_selected)) {
389 return;
390 }
391
392 if (useCudaPreprocessing(model_config_)) {
393 throw std::runtime_error(
394 "Failed to populate PBOD input tensors via CUDA preprocessing after "
395 "preprocessing.backend='cuda' was enabled");
396 }
397
398 populateModelInputOnCpu(point_features_host, pillar_ids_host, valid_mask_host, pillar_masks_host, num_selected);
399}

401void PBODModel::populateModelInputOnCpu(
402 float* point_features, std::int64_t* pillar_ids, bool* valid_mask, bool* pillar_masks, int num_selected) {
403 FeatureTensorView point_features_view(point_features, max_num_points_, preprocessed_feature_dim_);
404 Int64TensorView pillar_ids_view(pillar_ids, max_num_points_);
405 BoolTensorView valid_mask_view(valid_mask, max_num_points_);
406 BoolTensorView pillar_masks_view(pillar_masks, num_pillars_);
407
408 const float inv_voxel_x = 1.0F / voxel_x_;
409 const float inv_voxel_y = 1.0F / voxel_y_;
410 const int grid_y = static_cast<int>(model_config_.pillar_map_size[1]);
411 const float center_z = z_min_ + (z_max_ - z_min_) * 0.5F;
412
413 for (int pillar_id : active_pillar_ids_scratch_) {
414 const std::size_t sum_offset = static_cast<std::size_t>(pillar_id) * kPointCoordinateDim;
415 pillar_counts_[static_cast<std::size_t>(pillar_id)] = 0;
416 pillar_sum_[sum_offset + 0] = 0.0F;
417 pillar_sum_[sum_offset + 1] = 0.0F;
418 pillar_sum_[sum_offset + 2] = 0.0F;
419 pillar_sq_sum_[sum_offset + 0] = 0.0F;
420 pillar_sq_sum_[sum_offset + 1] = 0.0F;
421 pillar_sq_sum_[sum_offset + 2] = 0.0F;
422 pillar_mean_[sum_offset + 0] = 0.0F;
423 pillar_mean_[sum_offset + 1] = 0.0F;
424 pillar_mean_[sum_offset + 2] = 0.0F;
425 pillar_var_[sum_offset + 0] = 0.0F;
426 pillar_var_[sum_offset + 1] = 0.0F;
427 pillar_var_[sum_offset + 2] = 0.0F;
428 }
429 active_pillar_ids_scratch_.clear();
430
431 for (int i = 0; i < num_selected; ++i) {
432 const auto& point = selected_point_records_[static_cast<std::size_t>(i)];
433 const int ix = static_cast<int>((point.x - x_min_) * inv_voxel_x);
434 const int iy = static_cast<int>((point.y - y_min_) * inv_voxel_y);
435 const int pillar_id = ix * grid_y + iy;
436 pillar_ids_raw_[static_cast<std::size_t>(i)] = pillar_id;
437 pillar_ix_raw_[static_cast<std::size_t>(i)] = ix;
438 pillar_iy_raw_[static_cast<std::size_t>(i)] = iy;
439
440 const std::size_t count_offset = static_cast<std::size_t>(pillar_id);
441 if (pillar_counts_[count_offset] == 0) {
442 active_pillar_ids_scratch_.push_back(pillar_id);
443 pillar_masks_view(pillar_id) = true;
444 tensor_reset_state_.active_pillar_ids.push_back(pillar_id);
445 }
446 ++pillar_counts_[count_offset];
447 const std::size_t sum_offset = count_offset * kPointCoordinateDim;
448 pillar_sum_[sum_offset + 0] += point.x;
449 pillar_sum_[sum_offset + 1] += point.y;
450 pillar_sum_[sum_offset + 2] += point.z;
451 pillar_sq_sum_[sum_offset + 0] += point.x * point.x;
452 pillar_sq_sum_[sum_offset + 1] += point.y * point.y;
453 pillar_sq_sum_[sum_offset + 2] += point.z * point.z;
454 }
455
456 for (int pillar_id : active_pillar_ids_scratch_) {
457 const std::size_t count_offset = static_cast<std::size_t>(pillar_id);
458 const int32_t count = pillar_counts_[count_offset];
459 const std::size_t sum_offset = count_offset * kPointCoordinateDim;
460 const float inv_count = 1.0F / static_cast<float>(count);
461 const float mean_x = pillar_sum_[sum_offset + 0] * inv_count;
462 const float mean_y = pillar_sum_[sum_offset + 1] * inv_count;
463 const float mean_z = pillar_sum_[sum_offset + 2] * inv_count;
464 const float mean_x2 = pillar_sq_sum_[sum_offset + 0] * inv_count;
465 const float mean_y2 = pillar_sq_sum_[sum_offset + 1] * inv_count;
466 const float mean_z2 = pillar_sq_sum_[sum_offset + 2] * inv_count;
467 pillar_mean_[sum_offset + 0] = mean_x;
468 pillar_mean_[sum_offset + 1] = mean_y;
469 pillar_mean_[sum_offset + 2] = mean_z;
470 pillar_var_[sum_offset + 0] = std::max(mean_x2 - mean_x * mean_x, 0.0F);
471 pillar_var_[sum_offset + 1] = std::max(mean_y2 - mean_y * mean_y, 0.0F);
472 pillar_var_[sum_offset + 2] = std::max(mean_z2 - mean_z * mean_z, 0.0F);
473 }
474
475 tensor_reset_state_.active_point_rows.reserve(static_cast<std::size_t>(num_selected));
476 for (int i = 0; i < num_selected; ++i) {
477 const int pillar_id = pillar_ids_raw_[static_cast<std::size_t>(i)];
478 const std::size_t sum_offset = static_cast<std::size_t>(pillar_id) * kPointCoordinateDim;
479 const auto& point = selected_point_records_[static_cast<std::size_t>(i)];
480 const int ix = pillar_ix_raw_[static_cast<std::size_t>(i)];

518bool PBODModel::populateModelInputOnGpu(
519 float* point_features, std::int64_t* pillar_ids, bool* valid_mask, bool* pillar_masks, int num_selected) {
520 pcod_common::PillarPreprocessCudaConfig config;
521 config.x_min = x_min_;
522 config.x_max = x_max_;
523 config.y_min = y_min_;
524 config.y_max = y_max_;
525 config.z_min = z_min_;
526 config.z_max = z_max_;
527 config.voxel_x = voxel_x_;
528 config.voxel_y = voxel_y_;
529 config.value_threshold = value_threshold_;
530 config.min_value = min_value_;
531 config.max_value = max_value_;
532 config.epsilon = norm_epsilon_;
533 config.center_z = z_min_ + (z_max_ - z_min_) * 0.5F;
534 config.grid_x = static_cast<std::int32_t>(model_config_.pillar_map_size[0]);
535 config.grid_y = static_cast<std::int32_t>(model_config_.pillar_map_size[1]);
536 config.num_pillars = num_pillars_;
537 config.max_num_points = max_num_points_;
538 config.feature_dim = preprocessed_feature_dim_;
539 config.normalization_type = normalization_type_;
540 config.z_score_mean = 0.0F;
541 config.z_score_std = 1.0F;
542
543 // Recompute z-score stats directly for the CUDA config without changing
544 // point_preprocessor_ semantics.
545 if (normalization_type_ == pcod_common::PointFeatureNormalizationType::kZScore) {
546 double sum = 0.0;
547 double sum_sq = 0.0;
548 for (const auto& point : selected_point_records_) {
549 const double value = static_cast<double>(point.intensity);
550 sum += value;
551 sum_sq += value * value;
552 }
553 const double count = static_cast<double>(std::max(1, num_selected));
554 const double mean = sum / count;
555 const double variance = std::max(0.0, sum_sq / count - mean * mean);
556 config.z_score_mean = static_cast<float>(mean);
557 config.z_score_std = std::max(static_cast<float>(std::sqrt(variance)), norm_epsilon_);
558 }
559
560 const pcod_common::PillarPreprocessCudaOutputs outputs{point_features, pillar_ids, valid_mask, pillar_masks};
561 std::string error_message;
562 if (!cuda_preprocess_context_.run(selected_point_records_.data(), num_selected, config, outputs, &error_message)) {
563 RCLCPP_ERROR(rclcpp::get_logger("PBODModel"), "CUDA preprocessing failed: %s", error_message.c_str());
564 return false;
565 }
566
567 tensor_reset_state_.active_point_rows.resize(static_cast<std::size_t>(num_selected));
568 for (int i = 0; i < num_selected; ++i) {
569 tensor_reset_state_.active_point_rows[static_cast<std::size_t>(i)] = i;
570 }
571 tensor_reset_state_.active_pillar_ids.clear();
572 ConstBoolTensorView pillar_masks_view(pillar_masks, num_pillars_);
573 for (int pillar_id = 0; pillar_id < num_pillars_; ++pillar_id) {
574 if (pillar_masks_view(pillar_id)) {
575 tensor_reset_state_.active_pillar_ids.push_back(pillar_id);
576 }
577 }
578 active_pillar_ids_scratch_.clear();
579 return true;
580}

632std::vector<BoundingBox> PBODModel::modelOutputToBoxes() {
633 const int num_pillars = pillar_grid_.grid_x * pillar_grid_.grid_y;
634 const int num_classes = static_cast<int>(model_config_.predicted_class_names.size());
635 auto class_logits = triton_interface_.getOutputTensor<float>(kOutputNameClass, num_pillars, num_classes);
636 auto size_posterior = triton_interface_.getOutputTensor<float>(kOutputNameSize, num_pillars, kSizeValuesPerClass * num_classes);
637 auto focal_logits = triton_interface_.getOutputTensor<float>(kOutputNameFocal, num_pillars);
638 auto reg_logits =
639 triton_interface_.getOutputTensor<float>(kOutputNameReg, num_pillars, kRegressionValuesPerClass * num_classes);
640 const AuxiliaryGridMapRequest& auxiliary_grid_map_request = getAuxiliaryGridMapRequest();
641 if (has_auxiliary_grid_map_outputs_ && auxiliary_grid_map_request.any()) {
642 const auto build_auxiliary_grid_map = [&](const char* layer) {
643 return AuxiliaryGridMap{layer,
644 std::vector<float>(static_cast<std::size_t>(num_pillars)),
645 pillar_grid_.grid_x,
646 pillar_grid_.grid_y,
647 x_min_,
648 x_max_,
649 y_min_,
650 y_max_};
651 };
652
653 if (auxiliary_grid_map_request.density && auxiliary_grid_map_request.dynamic) {
654 auto density_logits = triton_interface_.getOutputTensor<float>(kOutputNameDensity, num_pillars);
655 auto dynamic_occupancy_logits = triton_interface_.getOutputTensor<float>(kOutputNameDynamicOccupancy, num_pillars);
656
657 density_grid_map_ = build_auxiliary_grid_map("density");
658 dynamic_grid_map_ = build_auxiliary_grid_map("dynamic");
659
660 for (int ix = 0; ix < pillar_grid_.grid_x; ++ix) {
661 for (int iy = 0; iy < pillar_grid_.grid_y; ++iy) {
662 const std::size_t flat_index = static_cast<std::size_t>(ix * pillar_grid_.grid_y + iy);
663 const Eigen::Index tensor_index = static_cast<Eigen::Index>(flat_index);
664 density_grid_map_->values[flat_index] = densityTransform(sigmoid(density_logits(tensor_index)));
665 dynamic_grid_map_->values[flat_index] = sigmoid(dynamic_occupancy_logits(tensor_index));
666 }
667 }
668 } else if (auxiliary_grid_map_request.density) {
669 auto density_logits = triton_interface_.getOutputTensor<float>(kOutputNameDensity, num_pillars);
670
671 density_grid_map_ = build_auxiliary_grid_map("density");
672 for (int ix = 0; ix < pillar_grid_.grid_x; ++ix) {
673 for (int iy = 0; iy < pillar_grid_.grid_y; ++iy) {
674 const std::size_t flat_index = static_cast<std::size_t>(ix * pillar_grid_.grid_y + iy);
675 density_grid_map_->values[flat_index] =
676 densityTransform(sigmoid(density_logits(static_cast<Eigen::Index>(flat_index))));
677 }
678 }
679 } else if (auxiliary_grid_map_request.dynamic) {
680 auto dynamic_occupancy_logits = triton_interface_.getOutputTensor<float>(kOutputNameDynamicOccupancy, num_pillars);
681
682 dynamic_grid_map_ = build_auxiliary_grid_map("dynamic");
683 for (int ix = 0; ix < pillar_grid_.grid_x; ++ix) {
684 for (int iy = 0; iy < pillar_grid_.grid_y; ++iy) {
685 const std::size_t flat_index = static_cast<std::size_t>(ix * pillar_grid_.grid_y + iy);
686 dynamic_grid_map_->values[flat_index] = sigmoid(dynamic_occupancy_logits(static_cast<Eigen::Index>(flat_index)));
687 }
688 }
689 }
690 }
691
692 pcod_common::PbodOutputsView view;
693 view.focal_logits = focal_logits.data();
694 view.size_posterior = size_posterior.data();
695 view.class_logits = class_logits.data();
696 view.reg_logits = reg_logits.data();
697 view.num_pillars = num_pillars;
698 view.num_classes = num_classes;
699 view.reg_dim = kRegressionValuesPerClass;
700
701 return pcod_common::DecodePbod(view, pillar_grid_, postprocess_config_);
702}
