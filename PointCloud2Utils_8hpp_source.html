29inline std::uint16_t byteSwap16(std::uint16_t value) { return static_cast<std::uint16_t>((value >> 8) | (value << 8)); }

34inline std::uint32_t byteSwap32(std::uint32_t value) {
35 return ((value & 0x000000FFU) << 24) | ((value & 0x0000FF00U) << 8) | ((value & 0x00FF0000U) >> 8) |
36 ((value & 0xFF000000U) >> 24);
37}

42inline std::uint64_t byteSwap64(std::uint64_t value) {
43 return ((value & 0x00000000000000FFULL) << 56) | ((value & 0x000000000000FF00ULL) << 40) |
44 ((value & 0x0000000000FF0000ULL) << 24) | ((value & 0x00000000FF000000ULL) << 8) |
45 ((value & 0x000000FF00000000ULL) >> 8) | ((value & 0x0000FF0000000000ULL) >> 24) |
46 ((value & 0x00FF000000000000ULL) >> 40) | ((value & 0xFF00000000000000ULL) >> 56);
47}

52template <typename T>
53T readScalarAt(const std::vector<std::uint8_t>& data, std::size_t offset) {
54 if (offset + sizeof(T) > data.size()) {
55 throw std::out_of_range("PointCloud2 field offset exceeds data buffer");
56 }
57 T value{};
58 std::memcpy(&value, &data[offset], sizeof(T));
59 return value;
60}

65inline float readFloat32At(const std::vector<std::uint8_t>& data, std::size_t offset, bool needs_swap) {
66 std::uint32_t bits = readScalarAt<std::uint32_t>(data, offset);
67 if (needs_swap) {
68 bits = byteSwap32(bits);
69 }
70 float out = 0.0F;
71 std::memcpy(&out, &bits, sizeof(out));
72 return out;
73}

91inline float readPointFieldAsFloat(const std::vector<std::uint8_t>& data,
92 std::size_t offset,
93 std::uint8_t datatype,
94 bool needs_swap) {
95 using PF = sensor_msgs::msg::PointField;
96 switch (datatype) {
97 case PF::FLOAT32:
98 return readFloat32At(data, offset, needs_swap);
99 case PF::FLOAT64:
100 return readFloat64AsFloatAt(data, offset, needs_swap);
101 case PF::UINT16: {
102 std::uint16_t value = readScalarAt<std::uint16_t>(data, offset);
103 if (needs_swap) {
104 value = byteSwap16(value);
105 }
106 return static_cast<float>(value);
107 }
108 case PF::UINT8:
109 return static_cast<float>(readScalarAt<std::uint8_t>(data, offset));
110 case PF::INT16: {
111 std::uint16_t raw = readScalarAt<std::uint16_t>(data, offset);
112 if (needs_swap) {
113 raw = byteSwap16(raw);
114 }
115 return static_cast<float>(static_cast<std::int16_t>(raw));
116 }
117 case PF::INT8:
118 return static_cast<float>(static_cast<std::int8_t>(readScalarAt<std::uint8_t>(data, offset)));
119 case PF::UINT32: {
120 std::uint32_t value = readScalarAt<std::uint32_t>(data, offset);
121 if (needs_swap) {
122 value = byteSwap32(value);
123 }
124 return static_cast<float>(value);
125 }
126 case PF::INT32: {
127 std::uint32_t raw = readScalarAt<std::uint32_t>(data, offset);
128 if (needs_swap) {
129 raw = byteSwap32(raw);
130 }
131 return static_cast<float>(static_cast<std::int32_t>(raw));
132 }
133 default:
134 throw std::runtime_error("Unsupported PointCloud2 datatype");
135 }
136}

141inline std::size_t pointCloud2PointOffset(const sensor_msgs::msg::PointCloud2& msg, std::size_t index) {
142 const std::size_t width = msg.width;
143 const std::size_t row = index / width;
144 const std::size_t col = index % width;
145 return row * static_cast<std::size_t>(msg.row_step) + col * static_cast<std::size_t>(msg.point_step);
146}

151inline PointCloud2PointRecord readPointCloud2PointRecord(const sensor_msgs::msg::PointCloud2& msg,
152 std::size_t index,
153 std::size_t x_offset,
154 std::size_t y_offset,
155 std::size_t z_offset,
156 std::size_t feature_offset,
157 std::uint8_t feature_datatype,
158 bool needs_swap) {
159 const std::size_t point_offset = pointCloud2PointOffset(msg, index);
160 return {readFloat32At(msg.data, point_offset + x_offset, needs_swap),
161 readFloat32At(msg.data, point_offset + y_offset, needs_swap),
162 readFloat32At(msg.data, point_offset + z_offset, needs_swap),
163 readPointFieldAsFloat(msg.data, point_offset + feature_offset, feature_datatype, needs_swap)};
164}
