63inline std::array<float, kPbodPreprocessedFeatureDim> makePbodFeatureRow(const PbodFeatureRowInputs& inputs) {
64 constexpr float kRadiusEpsilon = 1e-6F;
65 const float cluster_x = inputs.x - inputs.mean_x;
66 const float cluster_y = inputs.y - inputs.mean_y;
67 const float cluster_z = inputs.z - inputs.mean_z;
68 const float center_offset_x = inputs.x - inputs.center_x;
69 const float center_offset_y = inputs.y - inputs.center_y;
70 const float center_offset_z = inputs.z - inputs.center_z;
71 const float radius_squared = inputs.x * inputs.x + inputs.y * inputs.y;
72 const float radius = std::sqrt(radius_squared);
73 const float inverse_radius = 1.0F / (radius > kRadiusEpsilon ? radius : kRadiusEpsilon);
74 float sin_theta = 0.0F;
75 float cos_theta = 1.0F;
76 if (radius > 0.0F) {
77 const float inverse_norm = 1.0F / radius;
78 sin_theta = inputs.y * inverse_norm;
79 cos_theta = inputs.x * inverse_norm;
80 }
81
82 return {inputs.x,
83 inputs.y,
84 inputs.z,
85 radius,
86 inputs.z - inputs.z_min,
87 inverse_radius,
88 sin_theta,
89 cos_theta,
90 inputs.variance_x,
91 inputs.variance_y,
92 inputs.variance_z,
93 inputs.intensity,
94 cluster_x,
95 cluster_y,
96 cluster_z,
97 center_offset_x,
98 center_offset_y,
99 center_offset_z};
100}
