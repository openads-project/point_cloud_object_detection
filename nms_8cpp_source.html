31 explicit PreparedBox(const BoundingBox& box)
32 : minimum{std::numeric_limits<double>::infinity(), std::numeric_limits<double>::infinity()},
33 maximum{-std::numeric_limits<double>::infinity(), -std::numeric_limits<double>::infinity()},
34 bottom(box.z - box.height * 0.5),
35 top(box.z + box.height * 0.5),
36 area(static_cast<double>(box.length) * box.width),
37 has_height(box.height > 0.0f) {
38 const double c = std::cos(box.yaw), s = std::sin(box.yaw);
39 const double dx = box.length * 0.5, dy = box.width * 0.5;
40 vertices = {{{box.center[0] - dx * c + dy * s, box.center[1] - dx * s - dy * c},
41 {box.center[0] + dx * c + dy * s, box.center[1] + dx * s - dy * c},
42 {box.center[0] + dx * c - dy * s, box.center[1] + dx * s + dy * c},
43 {box.center[0] - dx * c - dy * s, box.center[1] - dx * s + dy * c}}};
44 for (const auto& p : vertices) {
45 for (std::size_t k = 0; k < 2; ++k) {
46 minimum[k] = std::min(minimum[k], p[k]);
47 maximum[k] = std::max(maximum[k], p[k]);
48 }
49 }
50 }
51};

59double HalfPlane(const Point& a, const Point& b, const Point& p) {
60 return (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0]);
61}

68double IntersectionArea(const PreparedBox& a, const PreparedBox& b) {
69 std::array<Point, 16> polygon{}, output{};
70 std::copy(a.vertices.begin(), a.vertices.end(), polygon.begin());
71 std::size_t count = 4;
72 for (std::size_t edge = 0; edge < 4 && count > 2; ++edge) {
73 const auto& start = b.vertices[edge];
74 const auto& end = b.vertices[(edge + 1) % 4];
75 std::size_t next_count = 0;
76 Point previous = polygon[count - 1];
77 double previous_distance = HalfPlane(start, end, previous);
78 for (std::size_t i = 0; i < count; ++i) {
79 const Point current = polygon[i];
80 const double distance = HalfPlane(start, end, current);
81 if ((distance >= 0.0) != (previous_distance >= 0.0)) {
82 const double t = previous_distance / (previous_distance - distance);
83 output[next_count++] = {previous[0] + t * (current[0] - previous[0]), previous[1] + t * (current[1] - previous[1])};
84 }
85 if (distance >= 0.0) {
86 output[next_count++] = current;
87 }
88 previous = current;
89 previous_distance = distance;
90 }
91 polygon.swap(output);
92 count = next_count;
93 }
94 if (count < 3) {
95 return 0.0;
96 }
97 double area = 0.0;
98 for (std::size_t i = 1; i + 1 < count; ++i) {
99 area += HalfPlane(polygon[0], polygon[i], polygon[i + 1]);
100 }
101 return std::abs(area) * 0.5;
102}

118PairOverlap EvaluateOverlap(
119 const PreparedBox& a, const PreparedBox& b, float nms_threshold, float merge_threshold, NmsStats& stats) {
120 double area_bound = std::min(a.area, b.area);
121 double rectangle_area = 1.0;
122 for (std::size_t k = 0; k < 2; ++k) {
123 const double extent = std::min(a.maximum[k], b.maximum[k]) - std::max(a.minimum[k], b.minimum[k]);
124 if (extent <= 0.0) {
125 return {};
126 }
127 rectangle_area *= extent;
128 }
129 area_bound = std::min(area_bound, rectangle_area);
130 double overlap_height = 1.0, volume_a = a.area, volume_b = b.area;
131 const bool has_height = a.has_height && b.has_height;
132 if (has_height) {
133 overlap_height = std::min(a.top, b.top) - std::max(a.bottom, b.bottom);
134 if (overlap_height <= 0.0) {
135 return {};
136 }
137 volume_a *= a.top - a.bottom;
138 volume_b *= b.top - b.bottom;
139 }
140 if (!(volume_a > 0.0 && volume_b > 0.0) || !std::isfinite(volume_a) || !std::isfinite(volume_b)) {
141 return {};
142 }
143 const double intersection_bound = area_bound * overlap_height;
144 const double merge_bound = static_cast<double>(merge_threshold) * merge_threshold * volume_a * volume_b;
145 const bool possible_suppression = intersection_bound > nms_threshold * (volume_a + volume_b - intersection_bound);
146 const bool possible_merge = has_height && intersection_bound * intersection_bound > merge_bound;
147 if (intersection_bound <= 0.0 || (!possible_suppression && !possible_merge)) {
148 return {};
149 }
150 ++stats.iou_evaluations;
151 const double intersection = std::min(IntersectionArea(a, b), area_bound) * overlap_height;
152 return {intersection / std::max(volume_a + volume_b - intersection, 1e-7),
153 has_height && intersection * intersection > merge_bound};
154}

160std::size_t BestClass(const BoundingBox& box) {
161 float best = -std::numeric_limits<float>::infinity();
162 std::size_t index = 0;
163 for (const auto& entry : box.classification) {
164 if (entry.score > best) {
165 best = entry.score;
166 index = entry.class_idx;
167 }
168 }

173struct ClassMixture {
174 std::vector<ClassificationEntry> probabilities;
175 double weight = 0.0;
176
181 void add(const BoundingBox& box, double score) {
182 if (box.classification.empty()) {
183 throw std::invalid_argument("Box merging requires class logits");
184 }
185 float maximum = -std::numeric_limits<float>::infinity();
186 for (const auto& entry : box.classification) {
187 if (std::isnan(entry.score) || entry.score == std::numeric_limits<float>::infinity()) {
188 throw std::invalid_argument("Box merging requires finite logits or negative infinity");
189 }
190 maximum = std::max(maximum, entry.score);
191 }
192 if (!std::isfinite(maximum)) {
193 throw std::invalid_argument("Box merging requires at least one finite class logit");
194 }
195 double denominator = 0.0;
196 for (const auto& entry : box.classification) {
197 denominator += std::exp(static_cast<double>(entry.score) - maximum);
198 }
199 for (const auto& entry : box.classification) {
200 auto existing = std::find_if(probabilities.begin(), probabilities.end(),
201 [&](const ClassificationEntry& value) { return value.class_idx == entry.class_idx; });
202 const float contribution = static_cast<float>(score * std::exp(static_cast<double>(entry.score) - maximum) / denominator);
203 if (existing == probabilities.end()) {
204 probabilities.push_back({entry.class_idx, contribution});
205 } else {
206 existing->score += contribution;
207 }
208 }
209 weight += score;
210 }
211
215 void finish(BoundingBox& box) {
216 box.classification = std::move(probabilities);
217 std::sort(box.classification.begin(), box.classification.end(),
218 [](const ClassificationEntry& a, const ClassificationEntry& b) { return a.class_idx < b.class_idx; });
219 for (auto& entry : box.classification) {
220 entry.score = static_cast<float>(std::log(entry.score / weight));
221 }
222 }
223};

237void ApplyRotatedNms(std::vector<BoundingBox>& bboxes, const NmsConfig& config, NmsStats* stats) {
238 NmsStats counts;
239 if (stats != nullptr) {
240 *stats = counts;
241 }
242 if (config.merge_enabled && (!std::isfinite(config.merge_score_threshold) || config.merge_score_threshold < 0.0f ||
243 config.merge_score_threshold > 1.0f)) {
244 throw std::invalid_argument("Box merging score threshold must be finite and in [0, 1]");
245 }
246 if (config.max_detections <= 0) {
247 bboxes.clear();
248 return;
249 }
250 std::vector<Candidate> scored;
251 scored.reserve(bboxes.size());
252 for (std::size_t i = 0; i < bboxes.size(); ++i) {
253 const auto& box = bboxes[i];
254 const std::size_t class_idx = BestClass(box);
255 const float threshold = config.score_thresholds.empty()
256 ? config.internal_score_threshold
257 : config.score_thresholds[std::min(class_idx, config.score_thresholds.size() - 1)];
258 const float score = box.detection_score.value_or(box.existence_probability);
259 if (std::isfinite(score) && score >= threshold) {
260 scored.push_back({score, i, class_idx, PreparedBox(box)});
261 }
262 }
263 counts.candidates = scored.size();
264 std::sort(scored.begin(), scored.end(), [](const Candidate& a, const Candidate& b) {
265 return a.score > b.score || (a.score == b.score && a.index < b.index);
266 });
267 std::vector<BoundingBox> kept;
268 std::vector<std::size_t> representatives;
269 std::vector<ClassMixture> mixtures;
270 const std::size_t capacity = std::min(scored.size(), static_cast<std::size_t>(config.max_detections));
271 kept.reserve(capacity);
272 representatives.reserve(capacity);
273 if (config.merge_enabled) {
274 mixtures.resize(capacity);
275 }
276 for (std::size_t i = 0; i < scored.size(); ++i) {
277 const auto& candidate = scored[i];
278 bool suppressed = false, merged = false;
279 for (std::size_t k = 0; k < kept.size(); ++k) {
280 const auto& representative = scored[representatives[k]];
281 const bool same_class = candidate.class_idx == representative.class_idx;
282 const bool can_merge = config.merge_enabled && candidate.geometry.has_height && representative.geometry.has_height;
283 if (!same_class && !can_merge) {
284 continue;
285 }
286 const auto overlap = EvaluateOverlap(candidate.geometry, representative.geometry, same_class ? config.iou_threshold : 1.0f,
287 can_merge ? config.merge_score_threshold : 1.0f, counts);
288 if (can_merge && overlap.merge) {
289 auto& mixture = mixtures[k];
290 // All-zero clusters use uniform weights. Once any positive score is
291 // present, zero-score members contribute no probability mass.
292 const double weight = representative.score > 0.0f ? std::max(0.0f, candidate.score) : 1.0;
293 if (mixture.probabilities.empty()) {
294 mixture.add(bboxes[representative.index], representative.score > 0.0f ? representative.score : 1.0);
295 }
296 mixture.add(bboxes[candidate.index], weight);
297 ++counts.merged_boxes;
298 merged = true;
299 break;
300 }
301 if (same_class && overlap.iou > config.iou_threshold) {
302 suppressed = true;
303 if (!config.merge_enabled) {
304 break;
305 }
306 }
307 }
308 if (!merged && !suppressed && kept.size() < capacity) {
309 kept.push_back(bboxes[candidate.index]);
310 representatives.push_back(i);
311 if (!config.merge_enabled && kept.size() == capacity) {
312 break;
313 }
314 }
315 }
316 for (std::size_t k = 0; k < kept.size() && config.merge_enabled; ++k) {
317 if (!mixtures[k].probabilities.empty()) {
318 mixtures[k].finish(kept[k]);
319 }
320 }
321 bboxes = std::move(kept);
322 if (stats != nullptr) {
323 *stats = counts;
324 }
325}
